cmake_minimum_required(VERSION 3.22)
project(reactor_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED REACTOR_FFI_DIR OR NOT DEFINED REACTOR_JNI_HEADERS)
  message(FATAL_ERROR "Set REACTOR_FFI_DIR and REACTOR_JNI_HEADERS to prebuilt inputs")
endif()
find_package(JNI REQUIRED)
add_library(reactor_jni SHARED src/abi.cpp src/client.cpp)
target_include_directories(reactor_jni PRIVATE ${JNI_INCLUDE_DIRS} include
  "${REACTOR_JNI_HEADERS}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../crates/reactor-ffi/include")
if(WIN32)
  target_link_libraries(reactor_jni PRIVATE "${REACTOR_FFI_DIR}/reactor_ffi.dll.lib")
  target_compile_options(reactor_jni PRIVATE /W4)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(reactor_jni PRIVATE "${REACTOR_FFI_DIR}/libreactor_ffi${CMAKE_SHARED_LIBRARY_SUFFIX}" Threads::Threads)
  target_compile_options(reactor_jni PRIVATE -Wall -Wextra -Werror)
  if(APPLE)
    set_target_properties(reactor_jni PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
  else()
    target_link_options(reactor_jni PRIVATE -static-libstdc++ -static-libgcc)
    set_target_properties(reactor_jni PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
  endif()
endif()
install(TARGETS reactor_jni LIBRARY DESTINATION . RUNTIME DESTINATION .)
